cmake_minimum_required(VERSION 3.21)
project(EveWorkbenchPrefixConsumer LANGUAGES CXX)
if(PRIOR_PACKAGE STREQUAL "Eve")
    find_package(Eve CONFIG REQUIRED COMPONENTS DesktopSdk PATHS "${PRIOR_PREFIX}" NO_DEFAULT_PATH)
else()
    find_package(ElaWidgetTools CONFIG REQUIRED PATHS "${PRIOR_PREFIX}/ElaWidgetTools/lib/cmake" NO_DEFAULT_PATH)
endif()
find_package(EveWorkbench CONFIG QUIET)
if(EveWorkbench_FOUND OR NOT EveWorkbench_NOT_FOUND_MESSAGE MATCHES "different SDK installation")
    message(FATAL_ERROR "EveWorkbench must reject an imported ${PRIOR_PACKAGE} from another prefix.")
endif()
unset(EveWorkbench_DIR CACHE)
find_package(EveWorkbench CONFIG REQUIRED PATHS "${PRIOR_PREFIX}" NO_DEFAULT_PATH)
find_package(EveWorkbench CONFIG REQUIRED)
