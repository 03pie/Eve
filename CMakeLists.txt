cmake_minimum_required(VERSION 3.21)
project(CordisCpp VERSION 0.1.0 LANGUAGES C CXX)

set(CORDIS_MSVC_SHOWINCLUDES_PREFIX "" CACHE STRING "Override a misdecoded MSVC /showIncludes prefix")
if(MSVC AND CORDIS_MSVC_SHOWINCLUDES_PREFIX)
    set(CMAKE_CL_SHOWINCLUDES_PREFIX "${CORDIS_MSVC_SHOWINCLUDES_PREFIX}")
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

option(CORDIS_BUILD_EXAMPLES "Build the foundation diagnostic example" ON)
option(CORDIS_WARNINGS_AS_ERRORS "Treat project warnings as errors" ON)
option(CORDIS_REQUIRE_QT5152 "Build Qt adapter and verify Qt 5.15.2 Core/Widgets/Test toolchain" OFF)
option(CORDIS_BUILD_ELA_UI "Build pinned Ela shell and page contribution services" OFF)
if(CORDIS_BUILD_ELA_UI AND NOT CORDIS_REQUIRE_QT5152)
    message(FATAL_ERROR "Ela UI requires CORDIS_REQUIRE_QT5152")
endif()

# 同工具链 Qt 插件与宿主必须使用同一 Core，不能在各 DLL 中复制身份/服务状态。
set(cordis_shared_core_default OFF)
if(WIN32 AND CORDIS_REQUIRE_QT5152 AND CMAKE_CXX_COMPILER_ID STREQUAL "MSVC")
    set(cordis_shared_core_default ON)
endif()
option(CORDIS_SHARED_CORE "Use one process Core DLL for same-toolchain C++ plugins" ${cordis_shared_core_default})
if(CORDIS_SHARED_CORE)
    if(NOT WIN32 OR NOT CMAKE_CXX_COMPILER_ID STREQUAL "MSVC")
        message(FATAL_ERROR "Shared Core currently supports the Windows/MSVC matrix only")
    endif()
    # Keep applications beside their shared runtime; per-fixture output overrides remain explicit.
    set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endif()

include(CTest)

if(CORDIS_REQUIRE_QT5152)
    find_package(Qt5 5.15.2 EXACT REQUIRED COMPONENTS Core Widgets Test)
    message(STATUS "Qt 5.15.2 found; enabling Cordis::Qt adapter and Qt tests")
endif()

function(cordis_target_defaults target)
    target_compile_features(${target} PUBLIC cxx_std_17)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
        if(CORDIS_WARNINGS_AS_ERRORS)
            target_compile_options(${target} PRIVATE /WX)
        endif()
    elseif(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
        if(CORDIS_WARNINGS_AS_ERRORS)
            target_compile_options(${target} PRIVATE -Werror)
        endif()
    endif()
endfunction()

add_subdirectory(src/core)
add_subdirectory(src/loader)
if(CORDIS_REQUIRE_QT5152)
    add_subdirectory(src/qt)
endif()
add_subdirectory(src/module)
if(CORDIS_BUILD_ELA_UI)
    add_subdirectory(src/ui)
endif()
if(TARGET cordis_module_host AND (CORDIS_BUILD_EXAMPLES OR BUILD_TESTING))
    add_subdirectory(examples/native_services)
endif()
if(CORDIS_BUILD_EXAMPLES)
    add_subdirectory(examples/builtin)
    if(CORDIS_REQUIRE_QT5152)
        add_subdirectory(examples/qt)
    endif()
endif()
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
