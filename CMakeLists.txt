project(Eve VERSION 0.3.1 LANGUAGES C CXX)
# The desktop product and the exact-toolchain business plugin SDK evolve independently.
set(EVE_SDK_VERSION 0.2.0)

set(EVE_MSVC_SHOWINCLUDES_PREFIX "" CACHE STRING "Override a misdecoded MSVC /showIncludes prefix")
if(MSVC AND EVE_MSVC_SHOWINCLUDES_PREFIX)
    set(CMAKE_CL_SHOWINCLUDES_PREFIX "${EVE_MSVC_SHOWINCLUDES_PREFIX}")
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

option(EVE_BUILD_EXAMPLES "Build the foundation diagnostic example" ON)
option(EVE_WARNINGS_AS_ERRORS "Treat project warnings as errors" ON)
option(EVE_REQUIRE_QT5152 "Build Qt adapter and verify Qt 5.15.2 Core/Widgets/Test toolchain" OFF)
option(EVE_BUILD_ELA_UI "Build pinned Ela shell and page contribution services" OFF)
if(EVE_BUILD_ELA_UI AND NOT EVE_REQUIRE_QT5152)
    message(FATAL_ERROR "Ela UI requires EVE_REQUIRE_QT5152")
endif()
if(WIN32 AND EVE_BUILD_ELA_UI)
    enable_language(RC)
endif()
include("${PROJECT_SOURCE_DIR}/cmake/EveBrand.cmake")

# 同工具链 Qt 插件与宿主必须使用同一 Core，不能在各 DLL 中复制身份/服务状态。
set(eve_shared_core_default OFF)
if(WIN32 AND EVE_REQUIRE_QT5152 AND CMAKE_CXX_COMPILER_ID STREQUAL "MSVC")
    set(eve_shared_core_default ON)
endif()
option(EVE_SHARED_CORE "Use shared Core/Qt/UI runtimes for same-toolchain C++ plugins" ${eve_shared_core_default})
if(EVE_SHARED_CORE)
    if(NOT WIN32 OR NOT CMAKE_CXX_COMPILER_ID STREQUAL "MSVC")
        message(FATAL_ERROR "Shared Core currently supports the Windows/MSVC matrix only")
    endif()
    # Keep applications beside their shared runtime; per-fixture output overrides remain explicit.
    set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endif()

include(CTest)

if(EVE_REQUIRE_QT5152)
    find_package(Qt5 5.15.2 EXACT REQUIRED COMPONENTS Core Widgets Test)
    message(STATUS "Qt 5.15.2 found; enabling Eve::Qt adapter and Qt tests")
endif()

function(eve_target_defaults target)
    target_compile_features(${target} PUBLIC cxx_std_17)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
        if(EVE_WARNINGS_AS_ERRORS)
            target_compile_options(${target} PRIVATE /WX)
        endif()
    elseif(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
        if(EVE_WARNINGS_AS_ERRORS)
            target_compile_options(${target} PRIVATE -Werror)
        endif()
    endif()
endfunction()
